```go
// Server is a Telegram RPC server.
type Server struct {
	// Configuration (set at creation, immutable after)
	transport     transport.Transport
	maxLayer      int
	interceptors  []Interceptor
	sessionStore  session.Store
	logger        Logger
	readTimeout   time.Duration
	writeTimeout  time.Duration
	
	// Layer support
	layerRegistry *layer.Registry
	
	// Crypto

	// Service registry
	registry *serviceRegistry
	
	// State
	mu         sync.RWMutex
	listeners  []net.Listener
	handlers   map[net.Conn]*connHandler
	closed     int32 // atomic
	shutdownCh chan struct{}
	
	// Metrics (optional)
	metrics *MetricsCollector
}

// NewServer creates a new TLRPC server with the provided options.
func NewServer(opts ...ServerOption) (*Server, error) {
	s := &Server{
		maxLayer:      222,
		readTimeout:   30 * time.Second,
		writeTimeout:  30 * time.Second,
		layerRegistry: layer.NewRegistry(),
		registry:      newServiceRegistry(),
		handlers:      make(map[net.Conn]*connHandler),
		shutdownCh:    make(chan struct{}),
		logger:        &defaultLogger{},
	}
	
	// Apply defaults and options
	for _, opt := range opts {
		if err := opt(s); err != nil {
			return nil, fmt.Errorf("apply option: %w", err)
		}
	}
	
	// Initialize defaults if not set
	if s.transport == nil {
		s.transport = transport.NewTCP()
	}
	
	if s.sessionStore == nil {
		s.sessionStore = session.NewMemoryStore()
	}
	
	s.cryptoManager = crypto.NewManager(s.sessionStore)
	
	// Register default layers if none specified
	if s.layerRegistry.Count() == 0 {
		for layer := 195; layer <= s.maxLayer; layer++ {
			// Layers registered by generated code later
		}
	}
	
	return s, nil
}

// Serve accepts incoming connections on the listener.
// Serve blocks until the server is stopped or an error occurs.
func (s *Server) Serve(lis net.Listener) error {
	s.mu.Lock()
	if s.isClosed() {
		s.mu.Unlock()
		return ErrServerClosed
	}
	s.listeners = append(s.listeners, lis)
	s.mu.Unlock()
	
	s.logger.Infof("tlrpc: server listening on %s", lis.Addr())
	
	for {
		select {
		case <-s.shutdownCh:
			return nil
		default:
		}
		
		conn, err := lis.Accept()
		if err != nil {
			if s.isClosed() {
				return nil
			}
			if ne, ok := err.(net.Error); ok && ne.Temporary() {
				s.logger.Warnf("tlrpc: accept error (temporary): %v", err)
				time.Sleep(5 * time.Millisecond)
				continue
			}
			return fmt.Errorf("accept: %w", err)
		}
		
		handler := s.newConnHandler(conn)
		
		s.mu.Lock()
		if s.isClosed() {
			s.mu.Unlock()
			conn.Close()
			return nil
		}
		s.handlers[conn] = handler
		s.mu.Unlock()
		
		go func() {
			defer func() {
				s.mu.Lock()
				delete(s.handlers, conn)
				s.mu.Unlock()
				conn.Close()
			}()
			
			if err := handler.run(); err != nil && !errors.Is(err, ErrConnClosed) {
				s.logger.Debugf("tlrpc: handler error: %v", err)
			}
		}()
	}
}

// Stop gracefully stops the server.
// It closes all listeners and waits for connections to close.
func (s *Server) Stop() error {
	if !atomic.CompareAndSwapInt32(&s.closed, 0, 1) {
		return ErrServerClosed
	}
	
	close(s.shutdownCh)
	
	s.mu.Lock()
	listeners := make([]net.Listener, len(s.listeners))
	copy(listeners, s.listeners)
	handlers := make(map[net.Conn]*connHandler, len(s.handlers))
	for k, v := range s.handlers {
		handlers[k] = v
	}
	s.mu.Unlock()
	
	// Close all listeners
	for _, lis := range listeners {
		lis.Close()
	}
	
	// Signal all handlers to stop
	for _, h := range handlers {
		h.stop()
	}
	
	// Wait with timeout
	done := make(chan struct{})
	go func() {
		for _, h := range handlers {
			h.wait()
		}
		close(done)
	}()
	
	select {
	case <-done:
		s.logger.Info("tlrpc: server stopped gracefully")
	case <-time.After(30 * time.Second):
		s.logger.Warn("tlrpc: server stop timed out")
	}
	
	return nil
}

// RegisterService registers a service implementation with the server.
// This is called by generated code (e.g., gen.RegisterAuthServer).
// Panics on invalid registration (grpc-style).
func (s *Server) RegisterService(desc ServiceDesc, impl interface{}) {
	if err := s.registry.register(desc, impl); err != nil {
		panic(err)
	}
}

func (s *Server) isClosed() bool {
	return atomic.LoadInt32(&s.closed) == 1
}

func (s *Server) newConnHandler(conn net.Conn) *connHandler {
	return &connHandler{
		server: s,
		conn:   conn,
		done:   make(chan struct{}),
	}
}
```

### 3.1 Interceptor Interface and Chain

**File**: `tlrpc/interceptor.go`

```go
package tlrpc

import (
	"context"
	"fmt"
	"runtime/debug"
	"time"
)

// Handler processes an RPC request.
// It receives the canonical request object and returns the response.
type Handler func(ctx context.Context, req interface{}) (interface{}, error)

// Interceptor intercepts RPC calls.
// It wraps a handler and can modify behavior.
type Interceptor func(next Handler) Handler

// Chain combines multiple interceptors into one.
// Interceptors are applied in order, so the first interceptor
// is the outermost (first to see request, last to see response).
func Chain(interceptors ...Interceptor) Interceptor {
	return func(final Handler) Handler {
		// Apply in reverse order so first is outermost
		for i := len(interceptors) - 1; i >= 0; i-- {
			final = interceptors[i](final)
		}
		return final
	}
}

// RecoveryInterceptor recovers from panics and returns errors.
func RecoveryInterceptor() Interceptor {
	return func(next Handler) Handler {
		return func(ctx context.Context, req interface{}) (resp interface{}, err error) {
			defer func() {
				if r := recover(); r != nil {
					err = fmt.Errorf("panic: %v\n%s", r, debug.Stack())
				}
			}()
			return next(ctx, req)
		}
	}
}

// LoggingInterceptor logs all RPC calls.
func LoggingInterceptor(logger Logger) Interceptor {
	return func(next Handler) Handler {
		return func(ctx context.Context, req interface{}) (interface{}, error) {
			start := time.Now()
			method := MethodFromContext(ctx)
			
			logger.Debugf("> %s %T", method, req)
			
			resp, err := next(ctx, req)
			
			duration := time.Since(start)
			if err != nil {
				logger.Debugf("< %s error=%v duration=%s", method, err, duration)
			} else {
				logger.Debugf("< %s %T duration=%s", method, resp, duration)
			}
			
			return resp, err
		}
	}
}

// AuthInterceptor enforces authentication for protected methods.
func AuthInterceptor(allowUnauthenticated ...string) Interceptor {
	allowed := make(map[string]bool)
	for _, m := range allowUnauthenticated {
		allowed[m] = true
	}
	
	return func(next Handler) Handler {
		return func(ctx context.Context, req interface{}) (interface{}, error) {
			method := MethodFromContext(ctx)
			
			if !allowed[method] && !IsAuthenticated(ctx) {
				return nil, ErrUnauthorized
			}
			
			return next(ctx, req)
		}
	}
}

// TimeoutInterceptor adds a timeout to the context.
func TimeoutInterceptor(timeout time.Duration) Interceptor {
	return func(next Handler) Handler {
		return func(ctx context.Context, req interface{}) (interface{}, error) {
			ctx, cancel := context.WithTimeout(ctx, timeout)
			defer cancel()
			return next(ctx, req)
		}
	}
}

// MetricsInterceptor records metrics for RPC calls.
func MetricsInterceptor(m *MetricsCollector) Interceptor {
	return func(next Handler) Handler {
		return func(ctx context.Context, req interface{}) (interface{}, error) {
			start := time.Now()
			method := MethodFromContext(ctx)
			
			m.RequestsTotal.WithLabelValues(method).Inc()
			m.RequestsInFlight.WithLabelValues(method).Inc()
			defer m.RequestsInFlight.WithLabelValues(method).Dec()
			
			resp, err := next(ctx, req)
			
			duration := time.Since(start).Seconds()
			m.RequestDuration.WithLabelValues(method).Observe(duration)
			
			if err != nil {
				m.ErrorsTotal.WithLabelValues(method, errorType(err)).Inc()
			}
			
			return resp, err
		}
	}
}

// Helper to extract method from context
func MethodFromContext(ctx context.Context) string {
	if m, ok := ctx.Value(methodKey).(string); ok {
		return m
	}
	return "unknown"
}

func errorType(err error) string {
	if err == nil {
		return "none"
	}
	// Classify error types...
	return "internal"
}

// Add method key to context package
const methodKey contextKey = 100
```

```go
package tlrpc

import (
	"errors"
	"fmt"
)

// Common errors
var (
	ErrServerClosed    = errors.New("tlrpc: server closed")
	ErrConnClosed      = errors.New("tlrpc: connection closed")
	ErrUnauthorized    = errors.New("tlrpc: unauthorized")
	ErrInvalidLayer    = errors.New("tlrpc: invalid layer version")
	ErrMethodNotFound  = errors.New("tlrpc: method not found")
	ErrInvalidRequest  = errors.New("tlrpc: invalid request")
	ErrTimeout         = errors.New("tlrpc: timeout")
	ErrNotImplemented  = errors.New("tlrpc: not implemented")
)

// RPCError represents a Telegram RPC error.
// These are sent to clients as error responses.
type RPCError struct {
	Code    int    // Error code (e.g., 401, 403, 500)
	Message string // Error message
	Type    string // Error type (e.g., "AUTH_KEY_UNREGISTERED")
}

func (e *RPCError) Error() string {
	return fmt.Sprintf("RPCError %d: %s (%s)", e.Code, e.Type, e.Message)
}

// Common RPC error constructors
func NewBadRequestError(msg string) *RPCError {
	return &RPCError{Code: 400, Type: "BAD_REQUEST", Message: msg}
}

func NewUnauthorizedError(msg string) *RPCError {
	return &RPCError{Code: 401, Type: "UNAUTHORIZED", Message: msg}
}

func NewForbiddenError(msg string) *RPCError {
	return &RPCError{Code: 403, Type: "FORBIDDEN", Message: msg}
}

func NewNotFoundError(msg string) *RPCError {
	return &RPCError{Code: 404, Type: "NOT_FOUND", Message: msg}
}

func NewInternalError(msg string) *RPCError {
	return &RPCError{Code: 500, Type: "INTERNAL", Message: msg}
}

// IsRPCError checks if an error is an RPCError.
func IsRPCError(err error) (*RPCError, bool) {
	var rpcErr *RPCError
	if errors.As(err, &rpcErr) {
		return rpcErr, true
	}
	return nil, false
}

// ConvertError converts a Go error to an RPCError.
// Non-RPC errors become internal errors.
func ConvertError(err error) *RPCError {
	if err == nil {
		return nil
	}
	
	if rpcErr, ok := IsRPCError(err); ok {
		return rpcErr
	}
	
	// Map common errors
	switch {
	case errors.Is(err, ErrUnauthorized):
		return NewUnauthorizedError("unauthorized")
	case errors.Is(err, ErrTimeout):
		return NewInternalError("timeout")
	default:
		return NewInternalError(err.Error())
	}
}

// Error codes from Telegram API
const (
	ErrCodeSeeOther       = 303
	ErrCodeBadRequest     = 400
	ErrCodeUnauthorized   = 401
	ErrCodeForbidden      = 403
	ErrCodeNotFound       = 404
	ErrCodeFlood          = 420
	ErrCodeInternal       = 500
)
```

### 5.1 Registry Implementation

**File**: `tlrpc/registry.go`

```go
package tlrpc

import (
	"fmt"
	"reflect"
	"sync"
)

// ServiceDesc describes a service.
// Generated code provides this.
type ServiceDesc struct {
	ServiceName string
	HandlerType interface{}
	Methods     []MethodDesc
}

// MethodDesc describes a service method.
type MethodDesc struct {
	MethodName string
	Handler    func(ctx context.Context, req interface{}) (interface{}, error)
}

// serviceRegistry stores registered services.
type serviceRegistry struct {
	mu       sync.RWMutex
	services map[string]*serviceInfo // by service name
	methods  map[string]MethodDesc   // by full method name
}

type serviceInfo struct {
	desc ServiceDesc
	impl interface{}
}

func newServiceRegistry() *serviceRegistry {
	return &serviceRegistry{
		services: make(map[string]*serviceInfo),
		methods:  make(map[string]MethodDesc),
	}
}

func (r *serviceRegistry) register(desc ServiceDesc, impl interface{}) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	
	if _, exists := r.services[desc.ServiceName]; exists {
		return fmt.Errorf("service %s already registered", desc.ServiceName)
	}
	
	// Validate implementation
	if err := r.validateImplementation(desc, impl); err != nil {
		return err
	}
	
	r.services[desc.ServiceName] = &serviceInfo{
		desc: desc,
		impl: impl,
	}
	
	// Index methods
	for _, m := range desc.Methods {
		fullName := desc.ServiceName + "." + m.MethodName
		if _, exists := r.methods[fullName]; exists {
			return fmt.Errorf("method %s already registered", fullName)
		}
		r.methods[fullName] = m
	}
	
	return nil
}

func (r *serviceRegistry) validateImplementation(desc ServiceDesc, impl interface{}) error {
	if impl == nil {
		return fmt.Errorf("implementation cannot be nil")
	}
	
	// Basic type check - full validation done at generation time
	implType := reflect.TypeOf(impl)
	if implType.Kind() != reflect.Ptr && implType.Kind() != reflect.Interface {
		return fmt.Errorf("implementation must be pointer or interface")
	}
	
	return nil
}

func (r *serviceRegistry) getMethod(name string) (MethodDesc, bool) {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	m, ok := r.methods[name]
	return m, ok
}

func (r *serviceRegistry) listServices() []string {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	names := make([]string, 0, len(r.services))
	for name := range r.services {
		names = append(names, name)
	}
	return names
}

func (r *serviceRegistry) listMethods() []string {
	r.mu.RLock()
	defer r.mu.RUnlock()
	
	names := make([]string, 0, len(r.methods))
	for name := range r.methods {
		names = append(names, name)
	}
	return names
}
```

### 6.1 Per-Connection Handler

**File**: `tlrpc/conn.go`

```go
package tlrpc

import (
	"context"
	"encoding/binary"
	"errors"
	"fmt"
	"io"
	"net"
	"sync"
	"sync/atomic"
	"time"

	"github.com/r6m/tlrpc/crypto"
	"github.com/r6m/tlrpc/mtproto"
	"github.com/r6m/tlrpc/session"
	"github.com/r6m/tlrpc/transport"
)

// connHandler handles a single client connection.
type connHandler struct {
	server *Server
	conn   transport.Conn
	
	// State
	authKey   crypto.AuthKey
	keyID     crypto.KeyID
	session   *session.Session
	layer     int
	
	// Lifecycle
	mu       sync.Mutex
	closed   int32
	done     chan struct{}
	wg       sync.WaitGroup
}

func (h *connHandler) run() error {
	defer close(h.done)
	
	// Set initial timeouts
	h.conn.SetDeadline(time.Now().Add(h.server.readTimeout))
	
	// Perform handshake
	if err := h.handshake(); err != nil {
		return fmt.Errorf("handshake: %w", err)
	}
	
	// Clear deadline after handshake
	h.conn.SetDeadline(time.Time{})
	
	// Main message loop
	for {
		if h.isClosed() {
			return ErrConnClosed
		}
		
		// Read message
		data, err := h.conn.ReadMessage()
		if err != nil {
			if errors.Is(err, io.EOF) || errors.Is(err, net.ErrClosed) {
				return nil // Clean close
			}
			return fmt.Errorf("read: %w", err)
		}
		
		// Process in goroutine for concurrency
		h.wg.Add(1)
		go func() {
			defer h.wg.Done()
			if err := h.handleMessage(data); err != nil {
				h.server.logger.Debugf("handle message: %v", err)
			}
		}()
	}
}

func (h *connHandler) handshake() error {
	// Read first message (should be unencrypted handshake)
	data, err := h.conn.ReadMessage()
	if err != nil {
		return err
	}
	
	// Parse as unencrypted message
	var msg mtproto.UnencryptedMessage
	if err := msg.Deserialize(data); err != nil {
		return fmt.Errorf("parse handshake: %w", err)
	}
	
	// Process handshake (req_pq_multi, etc.)
	// This is simplified - real handshake is complex
	key, err := h.server.cryptoManager.ProcessHandshake(&msg)
	if err != nil {
		return fmt.Errorf("process handshake: %w", err)
	}
	
	h.authKey = key
	h.keyID = key.ID()
	
	// Create or load session
	h.session, err = h.server.sessionStore.GetOrCreate(h.keyID)
	if err != nil {
		return fmt.Errorf("session: %w", err)
	}
	
	return nil
}

func (h *connHandler) handleMessage(data []byte) error {
	// Decrypt
	encrypted := &mtproto.EncryptedMessage{}
	if err := encrypted.Deserialize(data); err != nil {
		return fmt.Errorf("parse encrypted: %w", err)
	}
	
	inner, err := encrypted.Decrypt(h.authKey)
	if err != nil {
		return fmt.Errorf("decrypt: %w", err)
	}
	
	// Check message ID for duplicates
	if !h.session.IsValidMsgID(inner.MsgID) {
		return fmt.Errorf("invalid msg_id: %d", inner.MsgID)
	}
	h.session.AddMsgID(inner.MsgID)
	
	// Handle invokeWithLayer
	data = inner.Data
	if isInvokeWithLayer(data) {
		layer, innerData, err := parseInvokeWithLayer(data)
		if err != nil {
			return fmt.Errorf("parse invokeWithLayer: %w", err)
		}
		h.layer = min(layer, h.server.maxLayer)
		h.session.Layer = h.layer
		data = innerData
	}
	
	// Get layer handler
	layerHandler, ok := h.server.layerRegistry.Get(h.layer)
	if !ok {
		return ErrInvalidLayer
	}
	
	// Deserialize TL object
	obj, ctorID, err := layerHandler.Deserialize(data)
	if err != nil {
		return fmt.Errorf("deserialize: %w", err)
	}
	
	// Get method name
	method, ok := layerHandler.GetMethodName(obj)
	if !ok {
		return fmt.Errorf("unknown method for constructor %x", ctorID)
	}
	
	// Build context
	ctx := h.buildContext(method)
	
	// Find handler
	handler, ok := h.server.registry.getMethod(method)
	if !ok {
		return fmt.Errorf("method not found: %s", method)
	}
	
	// Apply interceptors
	chain := Chain(h.server.interceptors...)
	wrapped := chain(handler.Handler)
	
	// Execute
	resp, err := wrapped(ctx, obj)
	if err != nil {
		return h.sendError(inner.MsgID, err)
	}
	
	// Serialize response
	respData, err := layerHandler.Serialize(resp)
	if err != nil {
		return fmt.Errorf("serialize response: %w", err)
	}
	
	// Encrypt and send
	response := &mtproto.InnerData{
		MsgID: h.session.NextMsgID(),
		SeqNo: h.session.NextSeqNo(),
		Data:  respData,
	}
	
	encryptedResp, err := response.Encrypt(h.authKey, h.keyID)
	if err != nil {
		return fmt.Errorf("encrypt response: %w", err)
	}
	
	respBytes, err := encryptedResp.Serialize()
	if err != nil {
		return fmt.Errorf("serialize encrypted: %w", err)
	}
	
	return h.conn.WriteMessage(respBytes)
}

func (h *connHandler) buildContext(method string) context.Context {
	ctx := context.Background()
	ctx = NewContextWithSession(ctx, h.session)
	ctx = NewContextWithLayer(ctx, h.layer)
	ctx = NewContextWithAuthKeyID(ctx, h.keyID)
	ctx = NewContextWithUserID(ctx, h.session.UserID)
	ctx = context.WithValue(ctx, methodKey, method)
	return ctx
}
```
